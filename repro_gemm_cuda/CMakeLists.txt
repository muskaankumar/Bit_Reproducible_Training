cmake_minimum_required(VERSION 3.24)
project(repro_gemm_cuda LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES native)
find_package(CUDAToolkit REQUIRED)
add_library(repro_gemm repro_gemm.cu)
target_compile_options(repro_gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_include_directories(repro_gemm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_repro_gemm test_repro_gemm.cu)
target_link_libraries(test_repro_gemm PRIVATE repro_gemm)
add_executable(bench_repro_gemm bench_repro_gemm.cu)
target_link_libraries(bench_repro_gemm PRIVATE repro_gemm CUDA::cublas)
add_executable(hash_repro_gemm hash_repro_gemm.cu)
target_link_libraries(hash_repro_gemm PRIVATE repro_gemm)
